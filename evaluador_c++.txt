// Generador aleatorio moderno
std::mt19937 rng(std::random_device{}());

template <typename T>
T elegirAleatorio(const std::vector<T>& lista) {
    std::uniform_int_distribution<> dist(0, lista.size() - 1);
    return lista[dist(rng)];
}

// Funciones
PC generarPC() {
    PC pc;
    pc.cpu = elegirAleatorio(CPUs);
    pc.gpu = elegirAleatorio(GPUs);
    pc.ram = elegirAleatorio(RAMs).nivel;
    pc.almacenamiento = elegirAleatorio(storages);
    pc.so = elegirAleatorio(SOs);
    return pc;
}

std::string evaluarCompatibilidad(const PC& pc, const Juego& juego) {
    float rendimiento = 0.0;

    rendimiento += (float(pc.cpu.nivel) / juego.cpu_min.nivel) * juego.peso.cpu;
    rendimiento += (float(pc.gpu.nivel) / juego.gpu_min.nivel) * juego.peso.gpu;
    rendimiento += (float(pc.ram) / juego.ram_min) * juego.peso.ram;

    bool almacenamiento_ok = pc.almacenamiento >= juego.almacenamiento_min;

    bool so_ok = false;
    for (const auto& so : juego.sistemas_operativos) {
        if (pc.so == so) {
            so_ok = true;
            break;
        }
    }

    if (!almacenamiento_ok || !so_ok) return " No compatible";

    if (rendimiento >= 1.2) return " Rendimiento óptimo";
    else if (rendimiento >= 1.0) return " Rendimiento aceptable";
    else return " Rendimiento insuficiente";
}

void mostrarPC(const PC& pc) {
    std::cout << "\n PC Generada:\n";
    std::cout << "CPU: " << pc.cpu.nombre << "\n";
    std::cout << "GPU: " << pc.gpu.nombre << "\n";
    std::cout << "RAM: " << pc.ram << " GB\n";
    std::cout << "Almacenamiento: " << pc.almacenamiento << " GB\n";
    std::cout << "S.O.: " << pc.so << "\n\n";
}

// Nueva función: guarda la PC generada en un archivo TXT
void guardarPCenTXT(const PC& pc) {
    std::ofstream archivo("output/pc_generada.txt");
    if (!archivo) {
        std::cerr << "Error al crear el archivo de salida.\n";
        return;
    }

    archivo << "("
            << pc.cpu.nombre << ", "
            << pc.gpu.nombre << ", "
            << pc.ram << " GB, "
            << pc.almacenamiento << " GB, "
            << pc.so << ")";

    archivo.close();
    std::cout << "Archivo 'pc_generada.txt' creado exitosamente.\n\n";
}

void mostrarJuego(const Juego& juego) {

    std::cout << "Requisitos mínimos:\n";
    std::cout << "CPU: " << juego.cpu_min.nombre << "\n";
    std::cout << "GPU: " << juego.gpu_min.nombre << "\n";
    std::cout << "RAM: " << juego.ram_min << " GB\n";
    std::cout << "Almacenamiento: " << juego.almacenamiento_min << " GB\n";
    std::cout << "Sistemas Operativos compatibles: ";
    for (const auto& so : juego.sistemas_operativos) {
        std::cout << so << " ";
    }
    std::cout << "\n\n";
}

    } while (opcion != 3);
